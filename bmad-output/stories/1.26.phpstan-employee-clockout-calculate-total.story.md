```php
public function calculateTotalHours(): float
{
    if (empty($this->clock_in) || empty($this->clock_out)) {
        return 0.0;
    }

    $clockIn = $this->clock_in instanceof Carbon 
        ? $this->clock_in 
        : Carbon::parse($this->clock_in);
    $clockOut = $this->clock_out instanceof Carbon 
        ? $this->clock_out 
        : Carbon::parse($this->clock_out);

    $totalMinutes = $clockOut->diffInMinutes($clockIn);
    
    if ($this->break_duration > 0) {
        $totalMinutes -= $this->break_duration;
    }

    return max(0.0, round($totalMinutes / 60, 2));
}
```
